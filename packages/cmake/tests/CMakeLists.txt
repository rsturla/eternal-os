cmake_minimum_required(VERSION 3.20)
project(eternal_probe VERSION 1.0 LANGUAGES C)
include(CTest)
add_library(answer SHARED answer.c)
target_include_directories(answer PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(probe main.c)
target_link_libraries(probe PRIVATE answer)
add_test(NAME runtime COMMAND probe)
install(TARGETS answer probe RUNTIME DESTINATION bin LIBRARY DESTINATION lib)
